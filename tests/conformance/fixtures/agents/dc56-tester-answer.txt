    +// An enabled console's shutdown() removes its own listener and only that one.
    +
    +#include <unity.h>
    +#include <DomoticsCore/RemoteConsole.h>
    +
    +using namespace DomoticsCore;
    +using namespace DomoticsCore::Components;
    +
    +static int appLines = 0;
    +
    +void setUp(void) {
    +    HAL::Platform::setMillisForTest(1000);
    +    HAL::setCanOpenServerForTest(true);
    +}
    +
    +void tearDown(void) {
    +    HAL::Platform::resetMillisForTest();
    +}
    +
    +static int appLinesInConsole(RemoteConsoleComponent& console) {
    +    int n = 0;
    +    for (const auto& e : console.getRecentLogs(100)) {
    +        if (e.tag == "APP") n++;
    +    }
    +    return n;
    +}
    +
    +// Enabled console: Given an enabled console and an application listener added
    +// before it, When I run shutdown() and log a line, Then the console no longer
    +// receives the line and my own listener still does. Restart: Given a console
    +// shut down twice in a row, When I log a line, Then no other listener is
    +// removed.
    +void test_enabled_console_and_restart(void) {
    +    LoggerCallbacks::addCallback([](LogLevel, const char* tag, const char*) {
    +        if (String(tag) == "APP") appLines++;
    +    });
    +
    +    RemoteConsoleConfig config;
    +    config.enabled = true;
    +    RemoteConsoleComponent console(config);
    +    console.begin();
    +
    +    console.shutdown();
    +    DLOG_I("APP", "after the first shutdown");
    +    TEST_ASSERT_EQUAL_INT_MESSAGE(1, appLines, "the application listener was removed");
    +    TEST_ASSERT_EQUAL_INT_MESSAGE(0, appLinesInConsole(console),
        "the console still receives lines after shutdown");
    +
    +    console.shutdown();
    +    DLOG_I("APP", "after the second shutdown");
    +    TEST_ASSERT_EQUAL_INT_MESSAGE(2, appLines,
        "the second shutdown removed the application listener");
    +    TEST_ASSERT_EQUAL_INT_MESSAGE(0, appLinesInConsole(console),
        "the console received a line after the second shutdown");
    +}
    +
    +int main(int argc, char** argv) {
    +    UNITY_BEGIN();
    +    RUN_TEST(test_enabled_console_and_restart);
    +    return UNITY_END();
