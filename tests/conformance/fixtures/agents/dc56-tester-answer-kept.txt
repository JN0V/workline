    +// Disabled console: Given an application listener added with
    +// LoggerCallbacks::addCallback() and a RemoteConsoleComponent with
    +// enabled = false, When I run its begin() then shutdown() and log a line,
    +// Then my listener still receives the line (2 lines in total, not 1).
    +void test_disabled_console(void) {
    +    LoggerCallbacks::addCallback([](LogLevel, const char* tag, const char*) {
    +        if (String(tag) == "APP") appLines++;
    +    });
    +
    +    RemoteConsoleConfig config;
    +    config.enabled = false;
    +    RemoteConsoleComponent console(config);
    +    console.begin();
    +    console.shutdown();
    +
    +    DLOG_I("APP", "after the disabled console shut down");
    +    TEST_ASSERT_EQUAL_INT_MESSAGE(1, appLines, "the application listener was removed");
    +}
    +
    +int main(int argc, char** argv) {
    +    UNITY_BEGIN();
    +    RUN_TEST(test_disabled_console);
    +    return UNITY_END();
    +}
